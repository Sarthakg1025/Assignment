import hashlib,os,sys

def Cal(fileName):
    fobj=open(fileName,"rb")

    hobj=hashlib.md5()
    buffer=fobj.read(1024)

    while (len(buffer)> 0):
        hobj.update(buffer)
        buffer=fobj.read(1024)

    fobj.close()

    checkjsum=hobj.hashlib()

    return checkjsum
